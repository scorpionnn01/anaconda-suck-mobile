// Audio Synth & Horror BGM System
const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
let bgmInterval = null;

function startHorrorBGM() {
  if (bgmInterval) clearInterval(bgmInterval);
  if (audioCtx.state === 'suspended') audioCtx.resume();

  let beat = 0;
  bgmInterval = setInterval(() => {
    if (gameState !== 'PLAYING') return;
    
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    const droneNotes = [45, 48.5, 42, 38];
    
    osc.frequency.setValueAtTime(droneNotes[beat % droneNotes.length], audioCtx.currentTime);
    osc.type = 'sawtooth';
    
    gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.2);
    
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + 1.3);
    beat++;
  }, 900);
}

function stopBGM() {
  if (bgmInterval) {
    clearInterval(bgmInterval);
    bgmInterval = null;
  }
}

function playSound(type) {
  if (audioCtx.state === 'suspended') audioCtx.resume();
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  
  if (type === 'gun') {
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(160, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(20, audioCtx.currentTime + 0.09);
    gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.09);
  } else if (type === 'heavy') {
    osc.type = 'square';
    osc.frequency.setValueAtTime(200, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.14);
    gain.gain.setValueAtTime(0.35, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.14);
  } else if (type === 'bazooka') {
    osc.type = 'square';
    osc.frequency.setValueAtTime(100, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(10, audioCtx.currentTime + 0.3);
    gain.gain.setValueAtTime(0.45, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
  } else if (type === 'fireball') {
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(320, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.35);
    gain.gain.setValueAtTime(0.35, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.35);
  } else if (type === 'hit') {
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(220, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(30, audioCtx.currentTime + 0.12);
    gain.gain.setValueAtTime(0.35, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.12);
  } else if (type === 'scary_drone') {
    osc.type = 'sine';
    osc.frequency.setValueAtTime(140, audioCtx.currentTime);
    osc.frequency.linearRampToValueAtTime(30, audioCtx.currentTime + 1.8);
    gain.gain.setValueAtTime(0.6, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 1.8);
  }
  
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + 1.8);
}

// Game State
let gameState = 'MENU';
let currentLevel = 1;
let player = { x: width / 2, y: height - 80, hp: 100, maxHp: 100, speed: 10, fireCooldown: 0 };

let particles = [];

let anaconda = {
  x: width / 2, y: 60, hp: 350, maxHp: 350, speedY: 0.18, segments: [], length: 20,
  swirlAngle: 0, swirlRadius: 280, scaleSize: 22, spikes: 1, colorTheme: 'green',
  fireTimer: 0
};

// Infinite Weapons Progression with evolving gun flames and muzzle effects
function getWeaponDetails(lvl) {
  if (lvl === 1) return { name: 'AK-47', damage: 18, rate: 14, type: 'bullet', style: 'ak47', flameColor: '#ffcc00' };
  if (lvl === 2) return { name: 'Tactical Submachine Gun', damage: 28, rate: 11, type: 'bullet', style: 'smg', flameColor: '#00ffcc' };
  if (lvl === 3) return { name: 'Heavy Assault Rifle', damage: 45, rate: 9, type: 'bullet', style: 'assault', flameColor: '#ff5500' };
  if (lvl === 4) return { name: 'Quad-Barrel Gatling Gun', damage: 65, rate: 6, type: 'bullet', style: 'gatling', flameColor: '#00ffff' };
  if (lvl === 5) return { name: 'Standard Bazooka MK-I', damage: 120, rate: 22, type: 'missile', style: 'bazooka1', flameColor: '#ff2200' };
  if (lvl === 6) return { name: 'Advanced Plasma Launcher MK-II', damage: 180, rate: 20, type: 'missile', style: 'bazooka2', flameColor: '#bf00ff' };
  if (lvl === 7) return { name: 'Hyper-Velocity Rocket Cannon MK-III', damage: 260, rate: 18, type: 'missile', style: 'bazooka3', flameColor: '#ffff00' };
  if (lvl === 8) return { name: 'Quantum Destroyer Cannon MK-IV', damage: 370, rate: 16, type: 'missile', style: 'bazooka4', flameColor: '#00ff66' };
  if (lvl === 9) return { name: 'Apocalypse Titan Launcher MK-V', damage: 500, rate: 14, type: 'missile', style: 'bazooka5', flameColor: '#ff00aa' };
  
  let scaleFactor = lvl - 9;
  return { 
    name: `Infinite Doom Cannon MK-${9 + scaleFactor}`, 
    damage: 600 + scaleFactor * 150, 
    rate: Math.max(6, 13 - scaleFactor), 
    type: 'missile', 
    style: 'ultimate', 
    flameColor: `hsl(${ (lvl * 45) % 360 }, 100%, 60%)` 
  };
}

function initLevel(lvl) {
  currentLevel = lvl;
  player.x = width / 2;
  
  anaconda.x = width / 2;
  anaconda.y = 60;
  anaconda.swirlAngle = 0;
  
  let wpn = getWeaponDetails(lvl);
  
  anaconda.maxHp = 350 + (lvl - 1) * 260;
  anaconda.hp = anaconda.maxHp;
  anaconda.speedY = 0.2 + (lvl * 0.04); 
  
  anaconda.length = Math.min(30, 18 + Math.floor(lvl * 1.3));
  anaconda.scaleSize = Math.min(28, 18 + Math.floor(lvl * 0.7));
  
  // As level increases, swirl radius reduces drastically so movement becomes linear/straight down towards Suck
  let linearityFactor = Math.max(0.05, 1 - ((lvl - 1) * 0.1));
  anaconda.swirlRadius = Math.max(20, (width * 0.35 - (lvl * 15)) * linearityFactor);
  
  // Spikes and Dragon armor increment with each level
  anaconda.spikes = Math.min(6, 1 + Math.floor(lvl * 0.6));
  
  if (lvl <= 2) {
    anaconda.colorTheme = 'green';
  } else if (lvl <= 4) {
    anaconda.colorTheme = 'crimson';
  } else if (lvl <= 7) {
    anaconda.colorTheme = 'dragon_gold';
  } else {
    anaconda.colorTheme = 'nether_titan';
  }

  anaconda.segments = [];
  for (let i = 0; i < anaconda.length; i++) {
    anaconda.segments.push({ x: anaconda.x, y: anaconda.y - i * 14 });
  }

  document.getElementById('levelHUD').innerText = `LEVEL ${currentLevel}`;
  document.getElementById('weaponDisplay').innerText = wpn.name;
  document.getElementById('bossHpDisplay').innerText = anaconda.hp;
  document.getElementById('playerHpDisplay').innerText = player.hp;
}

// Multi-stage horror game over cinematic sequence
function gameOver() {
  gameState = 'GAMEOVER';
  stopBGM();
  playSound('scary_drone');

  const overlay = document.getElementById('overlay');
  const content = document.getElementById('overlayContent');
  const finger = document.getElementById('fingerIcon');
  const btnContainer = document.getElementById('menuBtnContainer');

  overlay.style.display = 'flex';
  finger.style.display = 'none';
  btnContainer.innerHTML = '';

  // Step 1: Show "YOU LOST" for 2.5 seconds
  content.innerHTML = `
    <h1 class="title-glow" style="font-size: 72px; letter-spacing: 6px;">YOU LOST</h1>
  `;

  setTimeout(() => {
    if (gameState !== 'GAMEOVER') return;
    // Step 2: Show Middle Finger and "Anaconda is the eternal villain of Sucks' life"
    finger.style.display = 'block';
    content.innerHTML = `
      <div class="subtitle-card" style="border-color: #ff3333; margin-top: 10px;">
        <p class="text-main" style="font-size: 26px;">Anaconda is the eternal villain of Sucks' life</p>
      </div>
    `;

    setTimeout(() => {
      if (gameState !== 'GAMEOVER') return;
      // Step 3: Fade to complete black for 1 second, then show Game Over and Replay/Exit options
      finger.style.display = 'none';
      content.innerHTML = '';
      overlay.style.background = '#000';

      setTimeout(() => {
        if (gameState !== 'GAMEOVER') return;
        overlay.style.background = 'radial-gradient(circle, rgba(20,0,0,0.96) 0%, rgba(0,0,0,0.99) 100%)';
        content.innerHTML = `
          <div style="font-size: 38px; color: #fff; text-shadow: 0 0 20px #ff0000; font-weight: bold; margin-bottom: 20px; letter-spacing: 5px; border-bottom: 2px solid #ff0000; padding-bottom: 8px;">GAME OVER</div>
        `;
        btnContainer.innerHTML = `
          <button class="actionBtn" onclick="restartGame()">REPLAY</button>
          <button class="actionBtn exitBtn" onclick="exitGame()">EXIT</button>
        `;
      }, 1000);

    }, 3000);

  }, 2500);
}

// Update Loop
function update() {
  if (gameState !== 'PLAYING') return;
  player.y = height - 80;

  if (keys['KeyA'] || keys['ArrowLeft']) player.x -= player.speed;
  if (keys['KeyD'] || keys['ArrowRight']) player.x += player.speed;
  
  if (player.x < 60) player.x = 60;
  if (player.x > width - 60) player.x = width - 60;

  let wpn = getWeaponDetails(currentLevel);

  if (player.fireCooldown > 0) player.fireCooldown--;
  if (keys['Space'] && player.fireCooldown <= 0) {
    if (wpn.type === 'missile') {
      playSound('bazooka');
      bullets.push({ x: player.x, y: player.y - 70, vx: 0, vy: -18, type: 'missile', damage: wpn.damage, flameColor: wpn.flameColor });
    } else {
      playSound(currentLevel >= 3 ? 'heavy' : 'gun');
      let spread = (Math.random() - 0.5) * 4;
      bullets.push({ x: player.x + spread, y: player.y - 70, vx: spread * 0.2, vy: -28, type: 'bullet', damage: wpn.damage, flameColor: wpn.flameColor });
    }

    for (let f = 0; f < 4; f++) {
      particles.push({
        x: player.x + (Math.random() - 0.5) * 18, y: player.y - 75,
        vx: (Math.random() - 0.5) * 7, vy: (Math.random() - 0.5) * 7 - 2,
        life: 8, color: wpn.flameColor, radius: Math.random() * 4 + 2
      });
    }
    player.fireCooldown = wpn.rate;
  }

  // Smooth transition from wavy/swirl to direct linear straight assault targeting Suck
  let linearityFactor = Math.max(0.05, 1 - ((currentLevel - 1) * 0.1));
  anaconda.swirlAngle += (0.015 + (currentLevel * 0.002)) * linearityFactor; 
  let targetX = (width / 2) + Math.sin(anaconda.swirlAngle) * anaconda.swirlRadius;
  
  // High-level linear tracking towards player X position as levels increase
  if (currentLevel >= 5) {
    targetX += (player.x - targetX) * Math.min(0.85, 0.2 + (currentLevel * 0.06));
  }

  anaconda.y += anaconda.speedY; 
  anaconda.x += (targetX - anaconda.x) * 0.08;

  if (currentLevel >= 5) {
    anaconda.fireTimer++;
    let fireInterval = Math.max(35, 100 - (currentLevel * 6));
    if (anaconda.fireTimer >= fireInterval) {
      anaconda.fireTimer = 0;
      playSound('fireball');
      let angle = Math.atan2((player.y - 40) - anaconda.y, player.x - anaconda.x);
      let fbSpeed = 6 + (currentLevel * 0.45);
      enemyFireballs.push({
        x: anaconda.x, y: anaconda.y,
        vx: Math.cos(angle) * fbSpeed, vy: Math.sin(angle) * fbSpeed,
        radius: 16, damage: 10 + Math.min(10, Math.floor(currentLevel * 1.5)) // Exactly 10 to 20 damage per flame hit as requested
      });
    }
  }

  anaconda.segments[0] = { x: anaconda.x, y: anaconda.y };
  for (let i = 1; i < anaconda.length; i++) {
    let segPrev = anaconda.segments[i - 1];
    let segCurr = anaconda.segments[i];
    let sdx = segPrev.x - segCurr.x;
    let sdy = segPrev.y - segCurr.y;
    let sdist = Math.hypot(sdx, sdy);
    let spacing = anaconda.scaleSize * 0.65;
    if (sdist > spacing) {
      segCurr.x = segPrev.x - (sdx / sdist) * spacing;
      segCurr.y = segPrev.y - (sdy / sdist) * spacing;
    }
  }

  if (anaconda.y >= height - 110 || Math.hypot(anaconda.x - player.x, anaconda.y - player.y) < (anaconda.scaleSize + 25)) {
    gameOver();
    return;
  }

  for (let i = enemyFireballs.length - 1; i >= 0; i--) {
    let fb = enemyFireballs[i];
    fb.x += fb.vx;
    fb.y += fb.vy;

    if (Math.hypot(fb.x - player.x, fb.y - (player.y - 30)) < 38) {
      player.hp -= fb.damage;
      playSound('hit');
      document.getElementById('playerHpDisplay').innerText = Math.max(0, player.hp);
      
      for (let p = 0; p < 10; p++) {
        particles.push({
          x: fb.x, y: fb.y, vx: (Math.random() - 0.5) * 9, vy: (Math.random() - 0.5) * 9,
          life: 12, color: '#ff2200', radius: Math.random() * 4 + 2
        });
      }

      enemyFireballs.splice(i, 1);
      if (player.hp <= 0) {
        gameOver();
        return;
      }
      continue;
    }

    if (fb.y > height || fb.x < 0 || fb.x > width) {
      enemyFireballs.splice(i, 1);
    }
  }

  for (let i = bullets.length - 1; i >= 0; i--) {
    let b = bullets[i];
    b.x += b.vx;
    b.y += b.vy;

    let hit = false;
    for (let j = 0; j < anaconda.segments.length; j++) {
      let seg = anaconda.segments[j];
      if (Math.hypot(b.x - seg.x, b.y - seg.y) < (anaconda.scaleSize + (b.type === 'missile' ? 18 : 0))) {
        anaconda.hp -= b.damage;
        document.getElementById('bossHpDisplay').innerText = Math.max(0, anaconda.hp);
        
        impactAnimations.push({
          x: b.x, y: b.y, radius: 4, maxRadius: b.type === 'missile' ? 55 : 28, alpha: 1.0, color: b.flameColor
        });
        
        for (let p = 0; p < (b.type === 'missile' ? 14 : 5); p++) {
          particles.push({
            x: b.x, y: b.y,
            vx: (Math.random() - 0.5) * 14, vy: (Math.random() - 0.5) * 14,
            life: 15, color: b.flameColor,
            radius: Math.random() * 4 + 2
          });
        }
        hit = true;
        break;
      }
    }

    if (hit) {
      bullets.splice(i, 1);
      if (anaconda.hp <= 0) {
        initLevel(currentLevel + 1);
      }
      continue;
    }

    if (b.y < 0 || b.x < 0 || b.x > width) bullets.splice(i, 1);
  }

  for (let i = impactAnimations.length - 1; i >= 0; i--) {
    let anim = impactAnimations[i];
    anim.radius += 2.5;
    anim.alpha -= 0.06;
    if (anim.alpha <= 0 || anim.radius >= anim.maxRadius) impactAnimations.splice(i, 1);
  }

  for (let i = particles.length - 1; i >= 0; i--) {
    let p = particles[i];
    p.x += p.vx;
    p.y += p.vy;
    p.life--;
    if (p.life <= 0) particles.splice(i, 1);
  }
}

function drawTempleBG() {
  let skyGrad = ctx.createLinearGradient(0, 0, 0, height);
  skyGrad.addColorStop(0, '#0f0502');
  skyGrad.addColorStop(0.5, '#4a1505');
  skyGrad.addColorStop(1, '#1a0802');
  ctx.fillStyle = skyGrad;
  ctx.fillRect(0, 0, width, height);

  ctx.fillStyle = 'rgba(255, 60, 0, 0.06)';
  for (let i = 0; i < 6; i++) {
    ctx.beginPath();
    ctx.moveTo(width * 0.16 * i, 0);
    ctx.lineTo(width * 0.16 * i + 150, height);
    ctx.lineTo(width * 0.16 * i + 40, height);
    ctx.fill();
  }

  ctx.fillStyle = '#261205';
  for (let i = 0; i <= width + 100; i += 220) {
    ctx.fillRect(i - 40, 0, 80, height);
    ctx.fillStyle = '#140a02';
    ctx.fillRect(i - 30, 0, 20, height);
    ctx.fillStyle = '#3d1d08';
    ctx.fillRect(i + 10, 0, 12, height);
    ctx.fillStyle = '#261205';
  }

  ctx.fillStyle = '#0d0502';
  ctx.fillRect(0, height - 90, width, 90);
  ctx.fillStyle = '#4a1a05';
  ctx.fillRect(0, height - 90, width, 10);
}

function draw() {
  drawTempleBG();

  if (gameState === 'PLAYING') {
    for (let i = anaconda.segments.length - 1; i >= 0; i--) {
      let seg = anaconda.segments[i];
      let radius = Math.max(12, anaconda.scaleSize - i * 0.25);

      ctx.save();
      ctx.translate(seg.x, seg.y);

      let mainColor = '#193d08', innerColor = '#327012';
      if (anaconda.colorTheme === 'crimson') {
        mainColor = '#4a0000'; innerColor = '#e60000';
      } else if (anaconda.colorTheme === 'dragon_gold') {
        mainColor = '#593800'; innerColor = '#ffbb00';
      } else if (anaconda.colorTheme === 'nether_titan') {
        mainColor = '#1a0033'; innerColor = '#ff00aa';
      }

      ctx.fillStyle = mainColor;
      ctx.beginPath();
      ctx.arc(0, 0, radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.strokeStyle = innerColor;
      ctx.lineWidth = 4;
      ctx.stroke();

      // Dragon-like spikes and scales scaling dynamically with level
      let spikeCount = anaconda.spikes;
      if (i < anaconda.segments.length * 0.9 && i % 2 === 0) {
        ctx.fillStyle = innerColor;
        for (let s = 0; s < spikeCount; s++) {
          let angleOffset = (s - (spikeCount - 1) / 2) * 0.45;
          ctx.save();
          ctx.rotate(angleOffset);
          ctx.beginPath();
          ctx.moveTo(-radius * 0.6, 0);
          ctx.lineTo(0, -radius - (12 + currentLevel * 2));
          ctx.lineTo(radius * 0.6, 0);
          ctx.fill();
          ctx.restore();
        }
      }

      if (i === 0) {
        // Dragon Eyes
        ctx.fillStyle = currentLevel >= 5 ? '#ffea00' : '#ff2200';
        ctx.shadowColor = '#ff0000';
        ctx.shadowBlur = 20;
        ctx.beginPath();
        ctx.arc(-radius * 0.45, -6, 6, 0, Math.PI * 2);
        ctx.arc(radius * 0.45, -6, 6, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;

        // Dragon Horns & Crests growing with level
        ctx.fillStyle = currentLevel >= 5 ? '#ff5500' : '#ffcc00';
        ctx.beginPath();
        ctx.moveTo(-radius * 0.4, -radius);
        ctx.lineTo(-radius * 0.95, -radius - (25 + currentLevel * 3));
        ctx.lineTo(-radius * 0.1, -radius);
        ctx.moveTo(radius * 0.4, -radius);
        ctx.lineTo(radius * 0.95, -radius - (25 + currentLevel * 3));
        ctx.lineTo(radius * 0.1, -radius);
        ctx.fill();

        // Dragon Fangs
        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.moveTo(-9, radius);
        ctx.lineTo(-5, radius + 14);
        ctx.lineTo(-2, radius);
        ctx.moveTo(2, radius);
        ctx.lineTo(5, radius + 14);
        ctx.lineTo(9, radius);
        ctx.fill();
      }

      ctx.restore();
    }

    enemyFireballs.forEach(fb => {
      ctx.save();
      ctx.fillStyle = currentLevel >= 5 ? '#ff3300' : '#ff5500';
      ctx.shadowColor = '#ff0000';
      ctx.shadowBlur = 20;
      ctx.beginPath();
      ctx.arc(fb.x, fb.y, fb.radius, 0, Math.PI * 2);
      ctx.fill();
      ctx.fillStyle = '#ffff00';
      ctx.beginPath();
      ctx.arc(fb.x, fb.y, fb.radius * 0.55, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;
      ctx.restore();
    });

    bullets.forEach(b => {
      ctx.save();
      if (b.type === 'missile') {
        ctx.fillStyle = b.flameColor;
        ctx.shadowColor = b.flameColor;
        ctx.shadowBlur = 12;
        ctx.fillRect(b.x - 6, b.y - 18, 12, 36);
        ctx.fillStyle = '#ffffff';
        ctx.fillRect(b.x - 2, b.y + 18, 4, 14);
      } else {
        ctx.fillStyle = b.flameColor;
        ctx.shadowColor = b.flameColor;
        ctx.shadowBlur = 10;
        ctx.fillRect(b.x - 2, b.y - 14, 5, 28);
      }
      ctx.restore();
    });

    impactAnimations.forEach(anim => {
      ctx.save();
      ctx.strokeStyle = anim.color;
      ctx.lineWidth = 3.5;
      ctx.globalAlpha = anim.alpha;
      ctx.beginPath();
      ctx.arc(anim.x, anim.y, anim.radius, 0, Math.PI * 2);
      ctx.stroke();
      ctx.restore();
    });

    particles.forEach(p => {
      ctx.fillStyle = p.color;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
      ctx.fill();
    });

    ctx.save();
    ctx.translate(player.x, player.y);

    let wpnStyle = getWeaponDetails(currentLevel).style;

    if (wpnStyle === 'ak47') {
      ctx.fillStyle = '#4a2f1d';
      ctx.fillRect(-18, 0, 10, 25);
      ctx.fillStyle = '#222';
      ctx.fillRect(-22, -25, 44, 18);
      ctx.fillStyle = '#5c3a21';
      ctx.fillRect(-16, -55, 10, 32);
      ctx.fillStyle = '#111';
      ctx.beginPath();
      ctx.moveTo(-4, -10);
      ctx.lineTo(-12, 15);
      ctx.lineTo(-2, 18);
      ctx.lineTo(2, -10);
      ctx.fill();
      ctx.fillStyle = '#1a1a1a';
      ctx.fillRect(-4, -75, 6, 52);
    } else if (wpnStyle === 'smg') {
      ctx.fillStyle = '#2b2b2b';
      ctx.fillRect(-18, -10, 36, 25);
      ctx.fillStyle = '#111';
      ctx.fillRect(-6, 0, 10, 30);
      ctx.fillRect(-3, -60, 6, 50);
    } else if (wpnStyle === 'assault') {
      ctx.fillStyle = '#333';
      ctx.fillRect(-24, -18, 48, 25);
      ctx.fillStyle = '#151515';
      ctx.fillRect(-8, 5, 12, 28);
      ctx.fillRect(-4, -68, 8, 52);
    } else if (wpnStyle === 'gatling') {
      ctx.fillStyle = '#1a1a1a';
      ctx.fillRect(-35, 5, 70, 30);
      ctx.fillStyle = '#333';
      ctx.fillRect(-25, -20, 50, 25);
      ctx.fillStyle = '#111';
      ctx.fillRect(-18, -65, 6, 48);
      ctx.fillRect(-6, -72, 6, 55);
      ctx.fillRect(6, -72, 6, 55);
      ctx.fillRect(12, -65, 6, 48);
    } else {
      ctx.fillStyle = '#3b4a22';
      ctx.fillRect(-20, -65, 40, 65);
      ctx.fillStyle = '#222';
      ctx.fillRect(-28, -35, 12, 8);
      ctx.fillRect(-8, 0, 16, 25);
      ctx.fillStyle = '#ff4400';
      ctx.fillRect(-6, -72, 12, 10);
    }

    ctx.restore();
  }

  requestAnimationFrame(() => {
    update();
    draw();
  });
}

draw();
